=== FILE: scripts/analyze_track.py ===
"""Phase 3: per-track measurement pipeline (machine measurements only, no listening claims).

Usage: python3 -I scripts/analyze_track.py <audio> <out_json> [<out_png>]

Every value written here is a signal measurement. Musical interpretation lives in the track reports.
Known traps handled:
  * beat trackers alias to 4/3, 3/2, 2x of the true tempo on swung hats -> tempo is chosen by a
    comb over onset autocorrelation at 4/8/16 beats, then refined by a least-squares beat grid.
  * peak onset energy in hip-hop sits on the snare (2 & 4) -> bar phase comes from 35-110 Hz kick energy.
"""
import sys, json, warnings
import numpy as np, soundfile as sf, librosa, scipy.signal as ss, pyloudnorm as pyln
warnings.filterwarnings('ignore')

KEYS = ['C', 'C#', 'D', 'D#', 'E', 'F', 'F#', 'G', 'G#', 'A', 'A#', 'B']
KK_MAJ = np.array([6.35, 2.23, 3.48, 2.33, 4.38, 4.09, 2.52, 5.19, 2.39, 3.66, 2.29, 2.88])
KK_MIN = np.array([6.33, 2.68, 3.52, 5.38, 2.60, 3.53, 2.54, 4.75, 3.98, 2.69, 3.34, 3.17])
TP_MAJ = np.array([0.748, 0.060, 0.488, 0.082, 0.670, 0.460, 0.096, 0.715, 0.104, 0.366, 0.057, 0.400])
TP_MIN = np.array([0.712, 0.084, 0.474, 0.618, 0.049, 0.460, 0.105, 0.747, 0.404, 0.067, 0.133, 0.330])
SR = 22050
HOP = 256
FPS = SR / HOP


def bp(y, lo, hi, sr=SR, order=4):
    if lo <= 0:
        sos = ss.butter(order, hi, 'lowpass', fs=sr, output='sos')
    elif hi >= sr / 2:
        sos = ss.butter(order, lo, 'highpass', fs=sr, output='sos')
    else:
        sos = ss.butter(order, [lo, hi], 'bandpass', fs=sr, output='sos')
    return ss.sosfiltfilt(sos, y)


def onset_env(y):
    S = np.abs(librosa.stft(y, n_fft=1024, hop_length=HOP))
    S = np.log1p(100 * S)
    d = np.maximum(0, np.diff(S, axis=1)).sum(0)
    return np.r_[0, d]


def key_scores(prof):
    out = []
    for name, maj, mn in (('kk', KK_MAJ, KK_MIN), ('temperley', TP_MAJ, TP_MIN)):
        for i in range(12):
            out.append((name, KEYS[i] + ' major', np.corrcoef(prof, np.roll(maj, i))[0, 1]))
            out.append((name, KEYS[i] + ' minor', np.corrcoef(prof, np.roll(mn, i))[0, 1]))
    return out


CHORD_T = {}
for i in range(12):
    for q, iv in (('', (0, 4, 7)), ('m', (0, 3, 7)), ('7', (0, 4, 7, 10)), ('m7', (0, 3, 7, 10)), ('maj7', (0, 4, 7, 11)),
                  ('dim', (0, 3, 6)), ('5', (0, 7))):
        v = np.zeros(12); v[[(i + k) % 12 for k in iv]] = 1
        CHORD_T[KEYS[i] + q] = v / np.linalg.norm(v)


def chord_of(c):
    c = c / (np.linalg.norm(c) + 1e-9)
    best = sorted(((float(c @ v), k) for k, v in CHORD_T.items()), reverse=True)
    return best[0][1], round(best[0][0], 2), best[1][1]


def main(path, out_json, out_png=None):
    x, sr0 = sf.read(path, always_2d=True)
    res = {'file': path.split('/')[-1], 'sr_source': sr0, 'dur_s': round(len(x) / sr0, 2)}
    # ---------- loudness / dynamics / stereo (source rate) ----------
    meter = pyln.Meter(sr0)
    res['lufs_integrated'] = round(meter.integrated_loudness(x), 2)
    mono0 = x.mean(1)
    res['sample_peak_dbfs'] = round(20 * np.log10(np.abs(x).max() + 1e-12), 2)
    rms = np.sqrt(np.mean(mono0 ** 2))
    res['crest_db'] = round(20 * np.log10(np.abs(mono0).max() / rms), 2)
    M, S_ = (x[:, 0] + x[:, -1]) / 2, (x[:, 0] - x[:, -1]) / 2
    res['side_minus_mid_db'] = round(10 * np.log10(np.mean(S_ ** 2) / np.mean(M ** 2) + 1e-12), 1)
    lowM = bp(M, 0, 150, sr0); lowS = bp(S_, 0, 150, sr0)
    res['side_minus_mid_below150_db'] = round(10 * np.log10(np.mean(lowS ** 2) / np.mean(lowM ** 2) + 1e-12), 1)
    # short-term loudness spread (3 s windows)
    win = int(3 * sr0); st = []
    for i in range(0, len(x) - win, win):
        l = meter.integrated_loudness(x[i:i + win]) if np.abs(x[i:i + win]).max() > 1e-4 else -70
        if l > -60: st.append(l)
    res['short_term_lufs_p10_p90'] = [round(float(np.percentile(st, 10)), 1), round(float(np.percentile(st, 90)), 1)] if st else None

    y = librosa.resample(mono0, orig_sr=sr0, target_sr=SR)
    # ---------- spectral balance ----------
    spec = np.abs(librosa.stft(y, n_fft=4096, hop_length=2048)) ** 2
    fr = librosa.fft_frequencies(sr=SR, n_fft=4096); tot = spec.sum()
    bands = [(20, 40), (40, 60), (60, 100), (100, 150), (150, 250), (250, 500), (500, 1000), (1000, 2000), (2000, 4000), (4000, 8000), (8000, 11025)]
    res['band_energy_pct'] = {f'{a}-{b}': round(100 * spec[(fr >= a) & (fr < b)].sum() / tot, 2) for a, b in bands}
    cen = librosa.feature.spectral_centroid(S=np.sqrt(spec), sr=SR)[0]
    res['spectral_centroid_hz_median'] = round(float(np.median(cen)), 0)
    # rolloff for lossy-source detection (where content dies)
    spec_db = 10 * np.log10(spec.mean(1) + 1e-20)
    ref = spec_db[(fr > 1000) & (fr < 4000)].mean()
    above = fr[spec_db > ref - 45]
    res['hf_cutoff_hz_est'] = int(above.max()) if len(above) else None

    # ---------- tempo: comb over bar-level autocorrelation ----------
    oe = onset_env(y); oe = oe - ss.medfilt(oe, 31); oe[oe < 0] = 0
    ac = np.correlate(oe, oe, 'full')[len(oe) - 1:]; ac /= ac[0] + 1e-9
    cands = []
    for bpm in np.arange(60, 180.01, 0.05):
        b = 60 / bpm * FPS
        s = sum(ac[int(round(k * b))] for k in (1, 2, 4, 8, 16) if int(round(k * b)) < len(ac))
        cands.append((s, bpm))
    cands.sort(reverse=True)
    top = []
    for s, b in cands:
        if all(abs(b - t[1]) > 1.5 for t in top): top.append((round(float(s), 3), round(float(b), 2)))
        if len(top) == 5: break
    # prefer the hip-hop lane (70-115) among near-equal candidates (within 8 %)
    best = top[0]
    for s, b in top:
        if 70 <= b <= 115 and s >= 0.92 * top[0][0]:
            best = (s, b); break
    bpm0 = best[1]
    res['tempo_candidates_comb'] = top
    # librosa tracker constrained near bpm0, then least-squares linear grid
    _, beats = librosa.beat.beat_track(onset_envelope=oe, sr=SR, hop_length=HOP, start_bpm=bpm0, tightness=400)
    bt = librosa.frames_to_time(beats, sr=SR, hop_length=HOP)
    per0 = 60 / bpm0
    # keep beats consistent with bpm0, fit t = off + k*per
    k = np.round((bt - bt[0]) / per0)
    A = np.vstack([k, np.ones_like(k)]).T
    per, off = np.linalg.lstsq(A, bt, rcond=None)[0]
    resid = bt - (off + k * per)
    # robust refit
    good = np.abs(resid) < 0.04
    per, off = np.linalg.lstsq(A[good], bt[good], rcond=None)[0]
    resid = bt[good] - (off + k[good] * per)
    res['bpm'] = round(60 / per, 3)
    res['bpm_fit_residual_ms_p90'] = round(float(np.percentile(np.abs(resid), 90) * 1000), 1)
    res['bpm_alternatives'] = {'half': round(30 / per, 2), 'double': round(120 / per, 2)}
    # grid drift test: per in first vs last third (tempo constant? drum machine vs live)
    n = len(bt[good]); kk = k[good]; tt = bt[good]
    if n > 30:
        p1 = np.polyfit(kk[:n // 3], tt[:n // 3], 1)[0]; p3 = np.polyfit(kk[-n // 3:], tt[-n // 3:], 1)[0]
        res['tempo_drift_pct_first_vs_last_third'] = round(100 * (p1 - p3) / p1, 3)

    # ---------- bar phase by kick ----------
    lo = np.abs(bp(y, 35, 110)); mid_sn = np.abs(bp(y, 150, 250)); hi_sn = np.abs(bp(y, 1500, 5000)); hat = np.abs(bp(y, 7000, 11000))
    nbeats = int((len(y) / SR - off) / per)
    beat_t = off + np.arange(nbeats) * per
    while beat_t[0] - per > 0: beat_t = np.r_[beat_t[0] - per, beat_t]

    def e_at(sig, t, w=0.07):
        i = int(t * SR); return float(sig[max(0, i - int(0.01 * SR)):i + int(w * SR)].mean()) if 0 <= i < len(sig) - 10 else 0.0

    kick_b = np.array([e_at(lo, t) for t in beat_t]); sn_b = np.array([e_at(hi_sn, t) for t in beat_t])
    phase_score = []
    for p in range(4):
        # beat 1 = strong kick, beats 2 & 4 = strong snare
        s1 = kick_b[p::4].mean(); s24 = sn_b[(p + 1) % 4::2].mean() / (sn_b.mean() + 1e-9)
        phase_score.append(s1 / (kick_b.mean() + 1e-9) + 0.5 * s24)
    ph = int(np.argmax(phase_score))
    srt = sorted(phase_score, reverse=True)
    res['bar_phase_margin'] = round(float(srt[0] - srt[1]), 3)
    bar_starts = beat_t[ph::4]
    res['first_downbeat_s'] = round(float(bar_starts[0]), 3)
    res['snare_backbeat_ratio'] = round(float(sn_b[(ph + 1) % 4::2].mean() / (sn_b[ph::2].mean() + 1e-9)), 2)
    barlen = 4 * per
    bar_starts = bar_starts[bar_starts + barlen <= len(y) / SR]
    res['n_bars'] = int(len(bar_starts)); res['bar_len_s'] = round(float(barlen), 4)

    # ---------- 16-step drum occupancy + microtiming ----------
    def band_onsets(sig_band):
        e = librosa.onset.onset_strength(y=sig_band, sr=SR, hop_length=128)
        pk, pr = ss.find_peaks(e, height=np.percentile(e, 80), distance=int(0.05 * SR / 128))
        return pk * 128 / SR, pr['peak_heights']

    yp = librosa.effects.percussive(y, margin=2.0)
    grids = {}
    micro = {}
    for name, (lo_f, hi_f) in {'kick': (35, 110), 'snare': (1500, 5000), 'hat': (7000, 11000)}.items():
        t_on, h_on = band_onsets(bp(yp, lo_f, hi_f))
        g = np.zeros(16); dev = [[] for _ in range(16)]
        step = barlen / 16
        for t, h in zip(t_on, h_on):
            j = np.searchsorted(bar_starts, t, side='right') - 1
            if j < 0 or j >= len(bar_starts): continue
            pos = (t - bar_starts[j]) / step
            s = int(np.round(pos)) % 16
            g[s] += h; dev[s].append((pos - np.round(pos)) * step * 1000)
        g = g / (g.max() + 1e-9)
        grids[name] = [round(float(v), 2) for v in g]
        micro[name] = [round(float(np.median(d)), 1) if len(d) > 4 else None for d in dev]
    res['drum_grid_16'] = grids
    res['drum_microtiming_ms_16'] = micro
    # swing: offbeat 16ths (steps 1,3,5..) delay vs straight, on hats
    hs = [micro['hat'][i] for i in (2, 6, 10, 14) if micro['hat'][i] is not None]
    h16 = [micro['hat'][i] for i in (1, 3, 5, 7, 9, 11, 13, 15) if micro['hat'][i] is not None]
    res['hat_8th_offbeat_delay_ms'] = round(float(np.median(hs)), 1) if hs else None
    res['hat_16th_offbeat_delay_ms'] = round(float(np.median(h16)), 1) if h16 else None
    sixteenth_ms = barlen / 16 * 1000
    if h16:
        res['swing_pct_16th'] = round(50 + 50 * float(np.median(h16)) / sixteenth_ms, 1)  # 50 = straight, 66.7 = triplet

    # ---------- per-bar features: layers, dropouts, chords ----------
    harm = librosa.effects.harmonic(y, margin=2.0)
    chroma = librosa.feature.chroma_cqt(y=harm, sr=SR, hop_length=512, bins_per_octave=36)
    cfps = SR / 512
    bass_chroma = librosa.feature.chroma_cqt(y=bp(harm, 0, 220), sr=SR, hop_length=512, fmin=librosa.note_to_hz('C1'), n_octaves=3)
    bands_bar = {'sub_35_110': lo, 'lowmid_150_250': mid_sn, 'mid_250_2k': np.abs(bp(y, 250, 2000)), 'presence_2k_5k': hi_sn, 'air_7k_11k': hat}
    harm_mid = np.abs(bp(harm, 250, 4000)); perc_all = np.abs(yp)
    bars = []
    for i, s in enumerate(bar_starts):
        a, b = int(s * SR), int((s + barlen) * SR)
        row = {'bar': i + 1, 't': round(float(s), 2)}
        row['rms_db'] = round(20 * np.log10(np.sqrt(np.mean(y[a:b] ** 2)) + 1e-9), 1)
        for k2, sig in bands_bar.items():
            row[k2] = round(20 * np.log10(np.mean(sig[a:b]) + 1e-9), 1)
        row['harm_mid_db'] = round(20 * np.log10(np.mean(harm_mid[a:b]) + 1e-9), 1)
        row['perc_db'] = round(20 * np.log10(np.mean(perc_all[a:b]) + 1e-9), 1)
        ca, cb = int(s * cfps), int((s + barlen) * cfps)
        cvec = chroma[:, ca:cb].mean(1); bvec = bass_chroma[:, ca:cb].mean(1)
        row['chroma'] = [round(float(v), 3) for v in cvec / (cvec.max() + 1e-9)]
        row['chord'], row['chord_fit'], row['chord_alt'] = chord_of(cvec)
        row['bass_pc'] = KEYS[int(np.argmax(bvec))]
        # half-bar chords
        hm = (ca + cb) // 2
        row['chord_half'] = [chord_of(chroma[:, ca:hm].mean(1))[0], chord_of(chroma[:, hm:cb].mean(1))[0]]
        row['bass_pc_beats'] = [KEYS[int(np.argmax(bass_chroma[:, int((s + q * per) * cfps):int((s + (q + 1) * per) * cfps)].mean(1)))] for q in range(4)]
        bars.append(row)
    res['bars'] = bars
    # dropouts: bars where sub drops >= 6 dB below track median (kick/bass removed)
    sub = np.array([r['sub_35_110'] for r in bars]); med = np.median(sub)
    perc = np.array([r['perc_db'] for r in bars]); pmed = np.median(perc)
    res['dropout_bars_sub'] = [r['bar'] for r in bars if r['sub_35_110'] < med - 6]
    res['dropout_bars_perc'] = [r['bar'] for r in bars if r['perc_db'] < pmed - 6]
    # harmonic loop length: chroma self-similarity at lags 1,2,4,8 bars
    C = np.array([r['chroma'] for r in bars]); C = C - C.mean(0); C = C / (np.linalg.norm(C, axis=1, keepdims=True) + 1e-9)  # mean-centred: similarity of the bar-to-bar deviations
    loop = {}
    for L in (1, 2, 3, 4, 8):
        if len(C) > L + 4: loop[L] = round(float(np.mean(np.sum(C[L:] * C[:-L], axis=1))), 3)
    res['chroma_bar_lag_similarity'] = loop
    # finer: beat-level chroma similarity at lags (beats) to find sub-bar loops
    # key
    prof = chroma.mean(1)
    ks = key_scores(prof)
    agg = {}
    for meth, kname, v in ks: agg.setdefault(kname, []).append(v)
    kr = sorted(((np.mean(v), k2) for k2, v in agg.items()), reverse=True)
    res['key_candidates'] = [(k2, round(float(v), 3)) for v, k2 in kr[:4]]
    res['key_confidence_margin'] = round(float(kr[0][0] - kr[1][0]), 3)
    res['pitch_class_profile'] = {KEYS[i]: round(float(prof[i] / prof.max()), 2) for i in range(12)}
    # bass pitch-class profile
    bp_prof = bass_chroma.mean(1)
    res['bass_pitch_class_profile'] = {KEYS[i]: round(float(bp_prof[i] / bp_prof.max()), 2) for i in range(12)}
    # structure: novelty on per-bar feature vectors
    F = np.array([[r['rms_db'], r['sub_35_110'], r['mid_250_2k'], r['presence_2k_5k'], r['air_7k_11k'], r['harm_mid_db'], r['perc_db']] + list(np.array(r['chroma']) * 3) for r in bars])
    Fz = (F - F.mean(0)) / (F.std(0) + 1e-9)
    nov = np.r_[0, np.linalg.norm(np.diff(Fz, axis=0), axis=1)]
    thr = max(np.percentile(nov, 88), np.median(nov) * 2)
    bounds = [0] + [i for i in range(1, len(nov)) if nov[i] >= thr and (i == 1 or nov[i] >= nov[i - 1])]
    secs = []
    for j, b0 in enumerate(bounds):
        b1 = bounds[j + 1] if j + 1 < len(bounds) else len(bars)
        seg = F[b0:b1]
        secs.append({'start_bar': b0 + 1, 'n_bars': b1 - b0, 't_start': bars[b0]['t'], 'rms_db': round(float(seg[:, 0].mean()), 1),
                     'sub': round(float(seg[:, 1].mean()), 1), 'mid': round(float(seg[:, 2].mean()), 1), 'air': round(float(seg[:, 4].mean()), 1),
                     'harm_mid': round(float(seg[:, 5].mean()), 1), 'perc': round(float(seg[:, 6].mean()), 1)})
    res['sections_auto'] = secs
    # transient sharpness of kick/snare: median attack slope proxy (peak / mean in 30 ms)
    res['perc_to_harm_ratio_db'] = round(float(20 * np.log10(np.sqrt(np.mean(yp ** 2)) / (np.sqrt(np.mean(harm ** 2)) + 1e-9))), 1)

    json.dump(res, open(out_json, 'w'), ensure_ascii=False, indent=1)
    if out_png:
        import matplotlib; matplotlib.use('Agg'); import matplotlib.pyplot as plt
        fig, ax = plt.subplots(4, 1, figsize=(16, 13), gridspec_kw={'height_ratios': [2, 2, 2, 1.4]})
        tb = [r['t'] for r in bars]
        for k2 in ['sub_35_110', 'mid_250_2k', 'presence_2k_5k', 'air_7k_11k', 'harm_mid_db', 'perc_db']:
            ax[0].plot(range(1, len(bars) + 1), [r[k2] for r in bars], label=k2)
        for s in secs: ax[0].axvline(s['start_bar'], color='k', alpha=.3)
        ax[0].legend(fontsize=7, ncol=6); ax[0].set_title(f"{res['file']}  |  {res['bpm']} BPM  |  key {res['key_candidates'][0][0]}  |  per-bar band energy (dB); lines = auto sections"); ax[0].set_xlabel('bar')
        ax[1].imshow(C.T, aspect='auto', origin='lower', extent=[0.5, len(bars) + .5, -.5, 11.5], cmap='magma'); ax[1].set_yticks(range(12)); ax[1].set_yticklabels(KEYS); ax[1].set_title('chroma per bar')
        S_db = librosa.amplitude_to_db(np.abs(librosa.stft(y[:int(min(len(y), (bar_starts[min(8, len(bar_starts) - 1)] + barlen) * SR))], n_fft=2048, hop_length=256)), ref=np.max)
        ax[2].imshow(S_db[:600], aspect='auto', origin='lower', cmap='magma', vmin=-80); ax[2].set_title('spectrogram, first ~8 bars (0-6.5 kHz)')
        G = np.array([grids['kick'], grids['snare'], grids['hat']])
        ax[3].imshow(G, aspect='auto', cmap='Greys'); ax[3].set_yticks([0, 1, 2]); ax[3].set_yticklabels(['kick', 'snare', 'hat']); ax[3].set_xticks(range(16)); ax[3].set_xticklabels(['1', 'e', '&', 'a', '2', 'e', '&', 'a', '3', 'e', '&', 'a', '4', 'e', '&', 'a']); ax[3].set_title('16-step onset occupancy (whole track)')
        plt.tight_layout(); plt.savefig(out_png, dpi=80); plt.close()


if __name__ == '__main__':
    main(*sys.argv[1:])

"""DSP building blocks: drum synthesis, vintage sampler emulation, saturation, EQ helpers, loudness."""
import numpy as np, scipy.signal as ss
from scipy.signal import resample_poly

SR = 48000


def _sos(kind, f, sr=SR, order=2):
    return ss.butter(order, f, kind, fs=sr, output='sos')


def hp(x, f, sr=SR, order=2): return ss.sosfilt(_sos('highpass', f, sr, order), x, axis=0)
def lp(x, f, sr=SR, order=2): return ss.sosfilt(_sos('lowpass', f, sr, order), x, axis=0)
def bpf(x, lo, hi, sr=SR, order=2): return ss.sosfilt(ss.butter(order, [lo, hi], 'bandpass', fs=sr, output='sos'), x, axis=0)


def peq(x, f0, gain_db, q=1.0, sr=SR):
    """RBJ peaking EQ."""
    A = 10 ** (gain_db / 40); w = 2 * np.pi * f0 / sr; al = np.sin(w) / (2 * q)
    b = [1 + al * A, -2 * np.cos(w), 1 - al * A]; a = [1 + al / A, -2 * np.cos(w), 1 - al / A]
    return ss.lfilter(np.array(b) / a[0], np.array(a) / a[0], x, axis=0)


def shelf(x, f0, gain_db, kind='low', sr=SR, s=1.0):
    A = 10 ** (gain_db / 40); w = 2 * np.pi * f0 / sr; cw, sw = np.cos(w), np.sin(w)
    al = sw / 2 * np.sqrt((A + 1 / A) * (1 / s - 1) + 2); sa = 2 * np.sqrt(A) * al
    if kind == 'low':
        b = [A * ((A + 1) - (A - 1) * cw + sa), 2 * A * ((A - 1) - (A + 1) * cw), A * ((A + 1) - (A - 1) * cw - sa)]
        a = [(A + 1) + (A - 1) * cw + sa, -2 * ((A - 1) + (A + 1) * cw), (A + 1) + (A - 1) * cw - sa]
    else:
        b = [A * ((A + 1) + (A - 1) * cw + sa), -2 * A * ((A - 1) + (A + 1) * cw), A * ((A + 1) + (A - 1) * cw - sa)]
        a = [(A + 1) - (A - 1) * cw + sa, 2 * ((A - 1) - (A + 1) * cw), (A + 1) - (A - 1) * cw - sa]
    return ss.lfilter(np.array(b) / a[0], np.array(a) / a[0], x, axis=0)


def sat(x, drive_db=6.0, mix=1.0):
    """tanh saturation, level-compensated."""
    g = 10 ** (drive_db / 20); y = np.tanh(x * g) / np.tanh(g)
    return mix * y + (1 - mix) * x


def env_exp(n, tau_s, sr=SR):
    return np.exp(-np.arange(n) / (tau_s * sr))


# ------------------------------------------------------------------ drum synthesis
def kick_808(f0=48.0, decay=0.9, pitch_drop=2.2, drop_tau=0.035, click=0.25, drive_db=4.0, sr=SR):
    """TR-808 style bass drum: sine with exponential pitch sweep + click, gently saturated."""
    n = int((decay * 3 + 0.05) * sr); t = np.arange(n) / sr
    f = f0 * (1 + (pitch_drop - 1) * np.exp(-t / drop_tau))
    ph = 2 * np.pi * np.cumsum(f) / sr
    y = np.sin(ph) * np.exp(-t / decay)
    ck = np.random.default_rng(1).standard_normal(n) * np.exp(-t / 0.0025)
    y += click * bpf(ck, 1000, 6000, sr)
    y = sat(y, drive_db)
    return (y / np.abs(y).max()).astype(np.float32)


def snare_synth(tone=185.0, noise_tau=0.11, tone_tau=0.06, snappy=0.7, sr=SR, seed=2):
    n = int(0.5 * sr); t = np.arange(n) / sr
    body = np.sin(2 * np.pi * tone * t) * np.exp(-t / tone_tau) + 0.5 * np.sin(2 * np.pi * tone * 1.6 * t) * np.exp(-t / (tone_tau * .7))
    nz = np.random.default_rng(seed).standard_normal(n) * np.exp(-t / noise_tau)
    nz = bpf(nz, 1800, 9000, sr)
    y = 0.6 * body + snappy * nz / (np.abs(nz).max() + 1e-9)
    return (y / np.abs(y).max()).astype(np.float32)


def hat_synth(open_=False, sr=SR, seed=3):
    n = int((0.45 if open_ else 0.08) * sr); t = np.arange(n) / sr
    # 808-style: six square oscillators, band-passed
    fr = [205.3, 304.4, 369.6, 522.7, 540.0, 800.0]
    y = sum(np.sign(np.sin(2 * np.pi * f * 2.0 * t + k)) for k, f in enumerate(fr))
    y = bpf(y, 7000, 14000, sr, 4) + 0.3 * hp(np.random.default_rng(seed).standard_normal(n), 8000, sr)
    y *= np.exp(-t / (0.12 if open_ else 0.018))
    return (y / np.abs(y).max()).astype(np.float32)


def clap_synth(sr=SR, seed=4):
    n = int(0.4 * sr); t = np.arange(n) / sr
    nz = bpf(np.random.default_rng(seed).standard_normal(n), 900, 5000, sr)
    env = np.zeros(n)
    for k, d in enumerate((0, 0.011, 0.022, 0.031)):
        i = int(d * sr); env[i:] += np.exp(-(t[:n - i]) / (0.008 if k < 3 else 0.09))
    y = nz * env
    return (y / np.abs(y).max()).astype(np.float32)


# ------------------------------------------------------------------ vintage samplers
def vintage(x, mode='sp1200', sr=SR):
    """Emulate the converter character of classic hip-hop samplers on a mono/stereo array.
    sp1200: 26.04 kHz, 12-bit, weak anti-alias (aliasing grit), output low-pass; the boom-bap crunch.
    mpc60 : 40 kHz, 12-bit companded-ish (16-bit store, 12-bit-like DAC colour), soft top.
    s950  : 39 kHz/31.25 kHz switchable filter, 12-bit, warm low-pass.
    """
    mono = x.ndim == 1
    X = x[:, None] if mono else x
    if mode == 'sp1200':
        fs2, bits, lpf = 26040, 12, 12500
    elif mode == 'mpc60':
        fs2, bits, lpf = 40000, 12, 16000
    elif mode == 's950':
        fs2, bits, lpf = 31250, 12, 10500
    else:
        return x
    # decimate with a weak (1st-order) pre-filter to keep some aliasing
    pre = lp(X, fs2 * 0.48, sr, 1)
    idx = np.arange(0, len(pre), sr / fs2)
    d = np.stack([np.interp(idx, np.arange(len(pre)), pre[:, c]) for c in range(X.shape[1])], 1)
    q = 2 ** (bits - 1)
    peak = np.abs(d).max() + 1e-9
    d = np.round(d / peak * q * 0.98) / q * peak / 0.98
    # zero-order-hold back up (imaging) then output filter
    back = resample_poly(d, sr, fs2, axis=0)
    back = lp(back, lpf, sr, 4)
    back = back[:len(X)]
    if len(back) < len(X): back = np.pad(back, ((0, len(X) - len(back)), (0, 0)))
    return back[:, 0] if mono else back


def pitch_resample(x, semitones):
    """Sampler-style pitch change (speed and pitch together, like an SP-1200/MPC)."""
    if abs(semitones) < 1e-3: return x
    r = 2 ** (semitones / 12)
    idx = np.arange(0, len(x) - 1, r)
    if x.ndim == 1: return np.interp(idx, np.arange(len(x)), x).astype(np.float32)
    return np.stack([np.interp(idx, np.arange(len(x)), x[:, c]) for c in range(x.shape[1])], 1).astype(np.float32)


# ------------------------------------------------------------------ dynamics
def compress(x, thr_db=-18, ratio=4, att_ms=10, rel_ms=120, makeup_db=0, knee_db=6, sidechain=None, sr=SR):
    """Feed-forward RMS-ish compressor (stereo-linked). sidechain: optional key signal."""
    key = x if sidechain is None else sidechain
    k = np.abs(key).max(axis=1) if key.ndim == 2 else np.abs(key)
    a = np.exp(-1 / (att_ms * 1e-3 * sr)); r = np.exp(-1 / (rel_ms * 1e-3 * sr))
    env = ss.lfilter([1 - r], [1, -r], k)  # release-shaped smoothing
    env = np.maximum(env, ss.lfilter([1 - a], [1, -a], k))
    lvl = 20 * np.log10(env + 1e-9)
    over = lvl - thr_db
    gr = np.where(over <= -knee_db / 2, 0, np.where(over >= knee_db / 2, over * (1 - 1 / ratio), (over + knee_db / 2) ** 2 / (2 * knee_db) * (1 - 1 / ratio)))
    g = 10 ** ((-gr + makeup_db) / 20)
    return x * (g[:, None] if x.ndim == 2 else g)


def soft_clip(x, thr=0.89):
    y = np.where(np.abs(x) < thr, x, np.sign(x) * (thr + (1 - thr) * np.tanh((np.abs(x) - thr) / (1 - thr))))
    return y


def true_peak_limit(x, ceiling_db=-1.0, lookahead_ms=1.5, rel_ms=60, sr=SR):
    """4x-oversampled peak detection, lookahead gain computer, delay-compensated."""
    c = 10 ** (ceiling_db / 20)
    up = resample_poly(x, 4, 1, axis=0)
    pk = np.abs(up).max(axis=1).reshape(-1, 4).max(axis=1) if len(up) % 4 == 0 else np.abs(up[:len(up) // 4 * 4]).max(axis=1).reshape(-1, 4).max(axis=1)
    pk = np.pad(pk, (0, len(x) - len(pk)), mode='edge')
    need = np.minimum(1, c / (pk + 1e-12))
    la = int(lookahead_ms * 1e-3 * sr)
    # min over lookahead window
    from scipy.ndimage import minimum_filter1d
    need = minimum_filter1d(need, size=2 * la + 1, origin=0)
    g = _release(need, np.exp(-1 / (rel_ms * 1e-3 * sr)))
    return x * g[:, None]


try:
    from numba import njit
except ImportError:  # pragma: no cover
    njit = lambda f: f


@njit
def _release(need, r):
    g = np.empty_like(need); cur = 1.0
    for i in range(len(need)):
        cur = need[i] if need[i] < cur else need[i] + (cur - need[i]) * r
        g[i] = cur
    return g


def lufs(x, sr=SR):
    import pyloudnorm as pyln
    return pyln.Meter(sr).integrated_loudness(x)


# ------------------------------------------------------------------ texture & synth voices
def vinyl(n, level_db=-38, crackle_per_s=9, sr=SR, seed=11):
    """Vinyl bed: band-limited hiss + sparse crackle impulses + slow rumble, stereo."""
    rng = np.random.default_rng(seed)
    hiss = bpf(rng.standard_normal((n, 2)), 1500, 9000, sr) * 0.25
    ck = np.zeros((n, 2))
    k = rng.poisson(crackle_per_s * n / sr)
    pos = rng.integers(0, n, k)
    ck[pos, rng.integers(0, 2, k)] = rng.uniform(-1, 1, k) * rng.pareto(3, k).clip(0, 6)
    ck = hp(ck, 1200, sr)
    rumble = lp(rng.standard_normal((n, 2)), 40, sr) * 0.6
    y = hiss + ck + rumble
    return (y / (np.abs(y).max() + 1e-9) * 10 ** (level_db / 20)).astype(np.float32)


def glide_lead(notes, spb, total_s, glide_s=0.07, vib_hz=5.2, vib_cents=18, sr=SR, bright=0.18):
    """G-funk whistle lead: one continuous oscillator (sine + a little saw), portamento between notes,
    delayed vibrato. notes = list of (beat, dur_beats, midi, vel)."""
    n = int(total_s * sr); f = np.zeros(n); a = np.zeros(n)
    notes = sorted(notes)
    prev = None
    for beat, dur, m, vel in notes:
        s, e = int(beat * spb * sr), int((beat + dur) * spb * sr)
        e = min(e, n)
        tgt = 440 * 2 ** ((m - 69) / 12)
        if prev is not None and s - prev[1] < int(0.05 * sr):
            g = min(int(glide_s * sr), e - s)
            f[s:s + g] = np.geomspace(prev[0], tgt, g); f[s + g:e] = tgt
        else:
            f[s:e] = tgt
        L = e - s; t = np.arange(L) / sr
        env = np.minimum(1, t / 0.012) * np.minimum(1, (L - np.arange(L)) / (0.06 * sr))
        a[s:e] = np.maximum(a[s:e], env * vel / 127)
        vib = np.clip((t - 0.18) / 0.25, 0, 1) * vib_cents / 1200
        f[s:e] *= 2 ** (vib * np.sin(2 * np.pi * vib_hz * t))
        prev = (tgt, e)
    f[f == 0] = 1
    ph = 2 * np.pi * np.cumsum(f) / sr
    saw = 2 * ((ph / (2 * np.pi)) % 1) - 1
    y = (np.sin(ph) * (1 - bright) + bright * lp(saw, 2500, sr)) * a
    return np.stack([y, y], 1).astype(np.float32)


def bass_808(notes, spb, total_s, decay=0.55, drive_db=7, glide_s=0.0, sr=SR):
    """Tuned 808 bass line: each note = sine with short pitch kick + saturation; notes=(beat,dur,midi,vel)."""
    n = int(total_s * sr); y = np.zeros(n)
    for beat, dur, m, vel in sorted(notes):
        s = int(beat * spb * sr); L = min(int((dur * spb + 0.08) * sr), n - s)
        if L <= 0: continue
        t = np.arange(L) / sr; f0 = 440 * 2 ** ((m - 69) / 12)
        f = f0 * (1 + 1.2 * np.exp(-t / 0.018))
        x = np.sin(2 * np.pi * np.cumsum(f) / sr) * np.exp(-t / decay) * (vel / 127)
        x *= np.minimum(1, (L - np.arange(L)) / (0.02 * sr))
        y[s:s + L] = x + y[s:s + L] * np.exp(-np.arange(L) / (0.004 * sr))  # choke previous note
    y = sat(y, drive_db)
    return np.stack([y, y], 1).astype(np.float32)


def width(x, w):
    """Mid/side width: w=1 unchanged, 0 = mono."""
    m = x.mean(1, keepdims=True); s = (x[:, :1] - x[:, 1:]) / 2
    return np.concatenate([m + w * s, m - w * s], 1)

"""Sampler for folders of single-note WAVs named like  spic_A2_v1_RR1.wav / susvib_C#3_v3_1.wav / KSHarp_D4_mf.wav (VSCO-2-CE)."""

from sfz import note_num

DYN = {'pp': 1, 'p': 2, 'mp': 3, 'mf': 4, 'f': 5, 'ff': 6}


class FolderSampler:
    def __init__(self, folder, sr=48000, release=0.25, gain_db=0.0, name_offset=12):
        """name_offset: VSCO-2-CE names most files one octave low (measured with pYIN: +12 st); harp folder is correct (0)."""

            m = re.search(r'_([A-Ga-g][#b]?-?\d)(?:_|\.)', b)
            if not m: continue

            v = re.search(r'_v(\d+)', b); d = re.search(r'_(pp|p|mp|mf|f|ff)(?:_|\.)', b)
            layer = int(v.group(1)) if v else DYN.get(d.group(1), 4) if d else 1
            self.map.setdefault(n, {}).setdefault(layer, []).append(p)
        self.notes = np.array(sorted(self.map)); self.cache = {}
        if not len(self.notes): raise ValueError('no samples in ' + folder)

    def _load(self, p):
        if p not in self.cache:
            x, sr = sf.read(p, always_2d=True, dtype='float32')
            if x.shape[1] == 1: x = np.repeat(x, 2, 1)
            if sr != self.sr: x = resample_poly(x, self.sr, sr, axis=0).astype(np.float32)
            self.cache[p] = x
        return self.cache[p]

    def render_note(self, key, vel, dur, rng=None):
        rng = rng or random
        src = int(self.notes[np.argmin(np.abs(self.notes - key))])
        layers = sorted(self.map[src]); li = layers[min(len(layers) - 1, int(vel / 128 * len(layers)))]
        x = self._load(rng.choice(self.map[src][li]))
        semis = key - src
        if semis:
            r = 2 ** (semis / 12); idx = np.arange(0, len(x) - 1, r)
            x = np.stack([np.interp(idx, np.arange(len(x)), x[:, c]) for c in range(2)], 1).astype(np.float32)

        return x * self.gain * (0.35 + 0.65 * (vel / 127) ** 1.6)

"""Read/write JUCE VST3 plugin state (MemoryBlock base64) so factory patches (.fxp) can be loaded headless via pedalboard."""
import re
TABLE = ".ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+"
IDX = {c: i for i, c in enumerate(TABLE)}


def juce_b64_decode(s):
    size, data = s.split('.', 1)
    size = int(size); out = bytearray(size); pos = 0
    for ch in data:
        v = IDX[ch]
        for b in range(6):
            if v >> b & 1:
                bit = pos + b
                if bit // 8 < size: out[bit // 8] |= 1 << (bit % 8)
        pos += 6
    return bytes(out)


def juce_b64_encode(b):
    nbits = len(b) * 8; chars = []
    for pos in range(0, nbits, 6):
        v = 0
        for k in range(6):
            bit = pos + k
            if bit < nbits and b[bit // 8] >> (bit % 8) & 1: v |= 1 << k
        chars.append(TABLE[v])
    return f"{len(b)}." + ''.join(chars)


def get_component(raw_state):
    xml = raw_state[8:].decode('utf-8', 'ignore') if raw_state[:4] == b'VC2!' else raw_state.decode('utf-8', 'ignore')
    m = re.search(r'<IComponent>([^<]+)</IComponent>', xml)
    return juce_b64_decode(m.group(1).strip())


def set_component(raw_state, comp_bytes):
    head = raw_state[:8]
    xml = raw_state[8:].decode('utf-8', 'ignore')
    xml = re.sub(r'<IComponent>[^<]+</IComponent>', '<IComponent>' + juce_b64_encode(comp_bytes) + '</IComponent>', xml)
    body = xml.encode('utf-8').rstrip(b'\x00')
    # VC2! header: magic + little-endian uint32 length of the xml payload (payload is NUL-terminated)
    import struct
    return b'VC2!' + struct.pack('<I', len(body)) + body + b'\x00'


def load_fxp_into(plugin, fxp_path):
    data = open(fxp_path, 'rb').read()
    i = data.find(b'sub3')
    plugin.raw_state = set_component(plugin.raw_state, data[i:])
    return plugin

"""Song model + renderer: tracks of note events -> stems -> mix bus -> master.

A Song is defined in beats. Swing and per-track micro-timing are applied at render time, so the same
pattern can be rendered straight, MPC-swung, or laid-back without touching the notes.
"""
import os, json, random
import numpy as np, soundfile as sf
import pedalboard
from mido import Message
import dsp
from dsp import SR

VST = '/usr/lib/vst3/'


class Note:
    __slots__ = ('beat', 'dur', 'pitch', 'vel')

    def __init__(self, beat, dur, pitch, vel=100):
        self.beat, self.dur, self.pitch, self.vel = beat, dur, pitch, vel


class Track:
    """instrument: ('sfz', SFZ) | ('vst', path, fxp_or_None, params) | ('oneshot', {pitch: array}) | ('func', f(pitch, vel, dur_s)->array)"""

    def __init__(self, name, instrument, notes=None, gain_db=0.0, pan=0.0, timing_ms=0.0, humanize_ms=0.0,
                 vel_jitter=0, swing=None, fx=None, send_reverb=0.0, mono=False):
        self.name, self.instrument = name, instrument
        self.notes = notes or []
        self.gain_db, self.pan, self.timing_ms, self.humanize_ms = gain_db, pan, timing_ms, humanize_ms
        self.vel_jitter, self.swing, self.fx, self.send_reverb, self.mono = vel_jitter, swing, fx, send_reverb, mono


class Song:
    def __init__(self, bpm, bars, swing=50.0, seed=7):
        self.bpm, self.bars, self.swing, self.seed = bpm, bars, swing, seed
        self.tracks = []

    @property
    def spb(self):  # seconds per beat
        return 60.0 / self.bpm

    def beat_to_s(self, beat, swing):
        """Swing delays every odd 16th; swing in % (50 = straight, 66.7 = triplet)."""
        six = beat * 4.0
        i = int(np.floor(six + 1e-6)); frac = six - i
        delay = 0.0
        if i % 2 == 1 and frac < 1e-3:
            delay = (swing - 50.0) / 50.0 * (self.spb / 4)
        return beat * self.spb + delay

    def length_s(self):
        return self.bars * 4 * self.spb


def _place(buf, x, start):
    if x is None or start >= len(buf): return
    if start < 0: x = x[-start:]; start = 0
    n = min(len(x), len(buf) - start)
    buf[start:start + n] += x[:n]


def render_track(song, tr, tail_s=4.0):
    rng = random.Random(song.seed + hash(tr.name) % 1000)
    n = int((song.length_s() + tail_s) * SR)
    buf = np.zeros((n, 2), np.float32)
    sw = tr.swing if tr.swing is not None else song.swing
    kind = tr.instrument[0]
    if kind in ('sfz', 'oneshot', 'func'):
        for nt in tr.notes:
            t = song.beat_to_s(nt.beat, sw) + tr.timing_ms / 1000 + rng.gauss(0, tr.humanize_ms / 1000 if tr.humanize_ms else 0)
            vel = int(np.clip(nt.vel + (rng.randint(-tr.vel_jitter, tr.vel_jitter) if tr.vel_jitter else 0), 1, 127))
            dur_s = nt.dur * song.spb
            if kind == 'sfz':
                x = tr.instrument[1].render_note(nt.pitch, vel, dur_s, rng)
            elif kind == 'oneshot':
                src = tr.instrument[1][nt.pitch]
                x = src * (vel / 127) ** 1.5
                if x.ndim == 1: x = np.stack([x, x], 1)
            else:
                x = tr.instrument[1](nt.pitch, vel, dur_s)
                if x.ndim == 1: x = np.stack([x, x], 1)
            if x is None:
                tr.missing = getattr(tr, 'missing', 0) + 1; continue
            _place(buf, x.astype(np.float32), int(t * SR))
    elif kind == 'vst':
        _, path, fxp, params = tr.instrument
        plug = pedalboard.load_plugin(VST + path)
        if fxp:
            import jucestate
            jucestate.load_fxp_into(plug, fxp)
            plug([], duration=0.5, sample_rate=SR, reset=False)
        for k, v in (params or {}).items():
            setattr(plug, k, v)
        msgs = []
        for nt in tr.notes:
            t = song.beat_to_s(nt.beat, sw) + tr.timing_ms / 1000 + rng.gauss(0, tr.humanize_ms / 1000 if tr.humanize_ms else 0)
            t = max(0.0, t)
            vel = int(np.clip(nt.vel + (rng.randint(-tr.vel_jitter, tr.vel_jitter) if tr.vel_jitter else 0), 1, 127))
            msgs.append((t, Message('note_on', note=int(nt.pitch), velocity=vel)))
            msgs.append((t + nt.dur * song.spb, Message('note_off', note=int(nt.pitch), velocity=0)))
        msgs.sort(key=lambda m: (m[0], m[1].type == 'note_on'))
        out = plug([m.copy(time=t) for t, m in msgs], duration=n / SR, sample_rate=SR, reset=False)
        buf = out.T.astype(np.float32)
        if buf.shape[1] == 1: buf = np.repeat(buf, 2, 1)
        buf = buf[:n]
    if tr.mono:
        m = buf.mean(1, keepdims=True); buf = np.repeat(m, 2, 1)
    if tr.fx is not None:
        buf = tr.fx(buf)
    # pan (constant power) + gain
    p = (tr.pan + 1) * np.pi / 4
    buf = buf * np.array([np.cos(p), np.sin(p)], np.float32) * np.sqrt(2) * 10 ** (tr.gain_db / 20)
    return buf


def plate_reverb(x, room=0.35, damp=0.6, wet=1.0):
    board = pedalboard.Pedalboard([pedalboard.Reverb(room_size=room, damping=damp, wet_level=wet, dry_level=0.0, width=0.9)])
    return board(x.T, SR).T


def mixdown(song, stems, reverb=None, bus=None, master=None, reverb_width=0.6):
    n = max(len(s) for s in stems.values())
    mix = np.zeros((n, 2), np.float32); send = np.zeros((n, 2), np.float32)
    for tr in song.tracks:
        s = stems[tr.name]
        mix[:len(s)] += s
        if tr.send_reverb > 0: send[:len(s)] += s * tr.send_reverb
    if send.any():
        rv = (reverb or plate_reverb)(send)
        rv = dsp.hp(rv, 250)  # keep the low end dry and mono-safe
        m_, s_ = rv.mean(1, keepdims=True), (rv[:, :1] - rv[:, 1:]) / 2
        rv = np.concatenate([m_ + reverb_width * s_, m_ - reverb_width * s_], 1)  # refs are narrow (median side-mid -19 dB)
        mix[:len(rv)] += rv
    if bus: mix = bus(mix)
    if master: mix = master(mix)
    return mix


def master_chain(x, target_lufs=-9.5, ceiling_db=-1.0, glue=True):
    y = x.copy()
    if glue:
        y = dsp.compress(y, thr_db=-14, ratio=2, att_ms=30, rel_ms=150, knee_db=6)
    # loudness: gain into soft clip + true-peak limiter, iterate to target
    g = 0.0
    for _ in range(6):
        z = dsp.true_peak_limit(dsp.soft_clip(y * 10 ** (g / 20)), ceiling_db)
        l = dsp.lufs(z)
        if abs(l - target_lufs) < 0.15: break
        g += (target_lufs - l)
    return z


def export(path_noext, x, mp3=True):

def auto_balance(stems, targets_lu, ref='snare'):
    """Gain-stage stems by measured loudness: targets_lu = {stem: LU relative to the reference stem}.
    Loudness = BS.1770 integrated (gated, so silence/rests don't count). Returns new dict + applied gains."""
    import pyloudnorm as pyln
    meter = pyln.Meter(SR)
    L = {k: meter.integrated_loudness(v) for k, v in stems.items() if np.abs(v).max() > 1e-5}
    base = L[ref]
    out, gains = {}, {}
    for k, v in stems.items():
        if k in targets_lu and k in L and np.isfinite(L[k]):
            g = base + targets_lu[k] - L[k]; out[k] = v * 10 ** (g / 20); gains[k] = round(g, 1)
        else:
            out[k] = v
    return out, gains

"""Minimal, dependency-light SFZ sampler (enough of SFZ v1/v2 for the sfzinstruments/Karoryfer libraries).

Supports: #define, #include, <control> default_path / set_cc, header inheritance (global > master > group > region),
key/lokey/hikey/pitch_keycenter, lovel/hivel, tune/transpose/pitch_keytrack, volume/amplitude/pan, offset,
seq_length/seq_position (round robin), lorand/hirand, trigger (attack only), sw_last/sw_default keyswitches,
locc/hicc conditions evaluated against set_cc defaults (+ overrides), loop_mode one_shot, ampeg_attack/release.
Not supported (ignored): CC modulation curves, filters, LFOs, crossfades, release triggers.
"""
import os, re, random
import numpy as np, soundfile as sf
from scipy.signal import resample_poly

NOTE_NAMES = {'c': 0, 'd': 2, 'e': 4, 'f': 5, 'g': 7, 'a': 9, 'b': 11}


def note_num(v):
    v = str(v).strip()
    try:
        return int(v)
    except ValueError:
        m = re.match(r'([a-gA-G])([#b]?)(-?\d+)', v)
        if not m: return None
        n = NOTE_NAMES[m.group(1).lower()] + (1 if m.group(2) == '#' else -1 if m.group(2) == 'b' else 0)
        return n + (int(m.group(3)) + 1) * 12


class SFZ:
    def __init__(self, path, cc=None, sr=48000, key_offset=0):
        """key_offset: measured correction so that render_note(k) SOUNDS at MIDI k (Growlybass: +12, measured with pYIN)."""
        self.key_offset = key_offset
        self.path = path; self.root = os.path.dirname(path); self.sr = sr
        self.defines = {}; self.cc = {}; self.default_path = ''
        self.regions = []; self.cache = {}; self.rr = {}
        text = self._load(path)
        self._parse(text)
        if cc: self.cc.update(cc)
        self.sw_default = None
        for r in self.regions:
            if 'sw_default' in r: self.sw_default = note_num(r['sw_default'])

    def _load(self, path, depth=0):
        txt = open(path, encoding='utf-8', errors='ignore').read()
        txt = re.sub(r'/\*.*?\*/', '', txt, flags=re.S)
        txt = re.sub(r'//[^\n]*', '', txt)
        def inc(m):
            p2 = os.path.join(self.root, m.group(1))
            return '\n' + self._load(p2, depth + 1) + '\n' if depth < 12 and os.path.exists(p2) else ''
        txt = re.sub(r'#include\s+"([^"]+)"', inc, txt)
        def dfn(m):
            self.defines[m.group(1)] = m.group(2); return ''
        txt = re.sub(r'#define\s+(\$\w+)\s+(\S+)', dfn, txt)
        return txt

    def _sub(self, s):
        for k in sorted(self.defines, key=len, reverse=True): s = s.replace(k, self.defines[k])
        return s

    def _parse(self, text):
        text = self._sub(text)
        tokens = re.split(r'(<\w+>)', text)
        level = {'global': {}, 'master': {}, 'group': {}}
        cur = None; cur_hdr = None
        def opcodes(s):
            # values may contain spaces (sample paths): split at ' key=' boundaries
            res = {}
            for m in re.finditer(r'(\w+)=(.*?)(?=\s+\w+=|$)', s.strip().replace('\n', ' ')):
                res[m.group(1)] = m.group(2).strip()
            return res
        for tok in tokens:
            if not tok.strip(): continue
            if re.match(r'<\w+>', tok):
                cur_hdr = tok[1:-1]
                if cur_hdr == 'global': level = {'global': {}, 'master': {}, 'group': {}}
                elif cur_hdr == 'master': level['master'] = {}; level['group'] = {}
                elif cur_hdr == 'group': level['group'] = {}
                continue
            ops = opcodes(tok)
            if cur_hdr == 'control':
                if 'default_path' in ops: self.default_path = ops['default_path'].replace('\\', '/')
                for k, v in ops.items():
                    m = re.match(r'set_(?:hd)?cc(\d+)', k)
                    if m:
                        val = float(v); self.cc[int(m.group(1))] = val * 127 if 'hdcc' in k else val
            elif cur_hdr in ('global', 'master', 'group'):
                level[cur_hdr].update(ops)
            elif cur_hdr == 'region':
                r = {}; r.update(level['global']); r.update(level['master']); r.update(level['group']); r.update(ops)
                if 'sample' in r and not r['sample'].startswith('*'):
                    r['_sample'] = os.path.normpath(os.path.join(self.root, self.default_path, r['sample'].replace('\\', '/')))
                    self.regions.append(r)

    # ------------------------------------------------------------------
    def _ok(self, r, key, vel):
        lk = note_num(r.get('lokey', r.get('key', 0))); hk = note_num(r.get('hikey', r.get('key', 127)))
        if lk is None or hk is None or not (lk <= key <= hk): return False
        if not (int(r.get('lovel', 1)) <= vel <= int(r.get('hivel', 127))): return False
        if r.get('trigger', 'attack') not in ('attack', 'first'): return False
        if 'sw_last' in r and self.sw_default is not None and note_num(r['sw_last']) != self.sw_default: return False
        for k, v in r.items():
            m = re.match(r'locc(\d+)$', k)
            if m and self.cc.get(int(m.group(1)), 0) < float(v): return False
            m = re.match(r'hicc(\d+)$', k)
            if m and self.cc.get(int(m.group(1)), 0) > float(v): return False
        return True

    def _sample(self, p):
        if p not in self.cache:
            if not os.path.exists(p):
                alt = [os.path.splitext(p)[0] + e for e in ('.flac', '.wav', '.ogg')]
                p2 = next((a for a in alt if os.path.exists(a)), None)
                if p2 is None: raise FileNotFoundError(p)
            else:
                p2 = p
            x, sr = sf.read(p2, always_2d=True, dtype='float32')
            if x.shape[1] == 1: x = np.repeat(x, 2, 1)
            if sr != self.sr: x = resample_poly(x, self.sr, sr, axis=0).astype(np.float32)
            self.cache[p] = x
        return self.cache[p]

    def render_note(self, key, vel, dur, rng=None):
        """Return stereo float32 array for one note (dur in seconds of note-on; release appended)."""
        rng = rng or random
        key = key + self.key_offset
        cands = [r for r in self.regions if self._ok(r, key, vel)]
        if not cands: return None
        # round robin groups
        if any('seq_length' in r for r in cands):
            slen = int(cands[0].get('seq_length', 1)); pos = self.rr.get((key, vel // 16), 0) % slen + 1
            self.rr[(key, vel // 16)] = pos
            c2 = [r for r in cands if int(r.get('seq_position', 1)) == pos]; cands = c2 or cands
        rv = rng.random()
        c2 = [r for r in cands if float(r.get('lorand', 0)) <= rv < float(r.get('hirand', 1.0001))]
        cands = c2 or cands
        out = None
        for r in cands:
            x = self._sample(r['_sample'])
            off = int(float(r.get('offset', 0)) * self.sr / 44100)
            x = x[off:]
            kc = note_num(r.get('pitch_keycenter', r.get('key', key))) or key
            kt = float(r.get('pitch_keytrack', 100)) / 100
            semis = (key - kc) * kt + float(r.get('transpose', 0)) + float(r.get('tune', 0)) / 100
            if abs(semis) > 1e-3:
                ratio = 2 ** (semis / 12)
                n = int(len(x) / ratio)
                idx = np.arange(n) * ratio
                x = np.stack([np.interp(idx, np.arange(len(x)), x[:, c]) for c in range(2)], 1).astype(np.float32)
            g = 10 ** (float(r.get('volume', 0)) / 20) * float(r.get('amplitude', 100)) / 100
            vt = float(r.get('amp_veltrack', 100)) / 100
            g *= (1 - vt) + vt * (vel / 127) ** 2
            pan = float(r.get('pan', 0)) / 100
            x = x * g * np.array([min(1, 1 - pan), min(1, 1 + pan)], dtype=np.float32)
            one_shot = r.get('loop_mode', '') == 'one_shot'
            if not one_shot:
                rel = float(r.get('ampeg_release', 0.25))
                n_on = int(dur * self.sr); n_rel = int(rel * self.sr)
                x = x[:n_on + n_rel].copy()
                if len(x) > n_on:
                    env = np.ones(len(x), dtype=np.float32)
                    tail = len(x) - n_on
                    env[n_on:] = np.exp(-5 * np.arange(tail) / max(1, n_rel))
                    x *= env[:, None]
            att = float(r.get('ampeg_attack', 0))
            if att > 0.0005:
                na = min(len(x), int(att * self.sr)); x[:na] *= np.linspace(0, 1, na)[:, None]
            out = x if out is None else _mix(out, x)
        return out


def _mix(a, b):

"""Pitch helpers."""
NAMES = {'C': 0, 'C#': 1, 'Db': 1, 'D': 2, 'D#': 3, 'Eb': 3, 'E': 4, 'F': 5, 'F#': 6, 'Gb': 6, 'G': 7, 'G#': 8, 'Ab': 8, 'A': 9, 'A#': 10, 'Bb': 10, 'B': 11}


def n(s):
    """'C4' -> 60, 'Gb3' -> 54"""
    name = s[:-1] if s[-1].isdigit() and not s[-2].isdigit() else s[:-2]
    octv = int(s[len(name):])
    return NAMES[name] + (octv + 1) * 12


def chord(root, quality):
    iv = {'maj': [0, 4, 7], 'min': [0, 3, 7], 'm7': [0, 3, 7, 10], 'maj7': [0, 4, 7, 11], '7': [0, 4, 7, 10], '9': [0, 4, 7, 10, 14],

          '7b9': [0, 4, 7, 10, 13], 'm11': [0, 3, 7, 10, 14, 17]}[quality]
    r = n(root) if isinstance(root, str) else root
    return [r + i for i in iv]

def render(song, tag):
    stems = {}
    for tr in song.tracks:
        if tr.instrument[0] == 'func' and tr.name in getattr(song, 'extra', {}):
            x = song.extra[tr.name]
            if tr.fx: x = tr.fx(x)
            p = (tr.pan + 1) * np.pi / 4
            stems[tr.name] = (x * np.array([np.cos(p), np.sin(p)]) * np.sqrt(2) * 10 ** (tr.gain_db / 20)).astype(np.float32)
            song._sends = getattr(song, '_sends', {})
        else:
            stems[tr.name] = render_track(song, tr)
    n_ = int((song.length_s() + 2) * SR)
    for k in stems:
        s = stems[k][:n_]
        stems[k] = np.pad(s, ((0, n_ - len(s)), (0, 0)))
    if getattr(song, 'balance', None):
        stems, gains = auto_balance(stems, song.balance)
        print(tag, 'auto-balance gains dB:', gains)
    mix = mixdown(song, stems)
    m = master_chain(mix, target_lufs=-10.0)
    export(OUT + tag, m)
    for k, v in stems.items():
        export(OUT + f'{tag}_stems/{k}', v, mp3=False)
    return m, stems


def metrics(x):
    mono = x.mean(1); spec = np.abs(np.fft.rfft(mono)) ** 2; f = np.fft.rfftfreq(len(mono), 1 / SR)
    M, S = (x[:, 0] + x[:, 1]) / 2, (x[:, 0] - x[:, 1]) / 2
    return dict(lufs=round(dsp.lufs(x), 2), true_peak_dbtp=round(20 * np.log10(np.abs(dsp.resample_poly(x, 4, 1, axis=0)).max()), 2),
                crest_db=round(20 * np.log10(np.abs(mono).max() / np.sqrt(np.mean(mono ** 2))), 2),
                low_le150_pct=round(100 * spec[f <= 150].sum() / spec.sum(), 1), mid_250_2k_pct=round(100 * spec[(f > 250) & (f < 2000)].sum() / spec.sum(), 1),
                centroid_hz=round(float((spec * f).sum() / spec.sum()), 0), side_minus_mid_db=round(10 * np.log10(np.mean(S ** 2) / np.mean(M ** 2) + 1e-12), 1),
                lr_correlation=round(float(np.corrcoef(x[:, 0], x[:, 1])[0, 1]), 3))

def layered_kick():
    """Sampled transient + knock + click, saturated. Peak-normalised."""
    rusty = SFZ(LIB + 'karoryfer.big-rusty-drums/Programs/02-basic.sfz')
    a = rusty.render_note(36, 127, 0.5)
    if a is None: a = C.kit('virt').render_note(36, 127, 0.5)
    b = C.kit('virt').render_note(36, 127, 0.5)
    L = max(len(a), len(b)); k = np.zeros((L, 2), np.float32)
    k[:len(a)] += dsp.bpf(a, 55, 220, order=2) * 1.2            # knock
    k[:len(b)] += dsp.hp(b, 1500, order=2) * 0.9                  # click/beater
    # synthetic thump glues the two (60 Hz, 120 ms)
    t = np.arange(int(0.25 * SR)) / SR; th = np.sin(2 * np.pi * (54 * (1 + 1.5 * np.exp(-t / 0.012))).cumsum() / SR) * np.exp(-t / 0.07)
    k[:len(t)] += np.stack([th, th], 1) * 0.55
    k = dsp.peq(k, 75, 3, 1.0); k = dsp.peq(k, 3200, 4, 1.2); k = dsp.sat(k, 7)
    k = k[:int(0.45 * SR)] * np.minimum(1, (len(k[:int(0.45 * SR)]) - np.arange(len(k[:int(0.45 * SR)]))) / (0.03 * SR))[:, None]
    return (k / np.abs(k).max()).astype(np.float32)


def layered_snare():

def bass_808_v4(notes, spb, total_s, decay=0.75, drive_db=7, sr=SR):
    """notes = (beat, dur, midi, vel[, glide_from_midi]). Sine + pitch kick + glide + drive; 2nd-harmonic presence for small speakers."""
    N = int(total_s * sr); y = np.zeros(N)
    for nt in sorted(notes):
        beat, dur, m, vel = nt[:4]; gfrom = nt[4] if len(nt) > 4 else None
        s = int(beat * spb * sr); L = min(int((dur * spb + 0.15) * sr), N - s)
        if L <= 0: continue
        t = np.arange(L) / sr; f0 = 440 * 2 ** ((m - 69) / 12)
        f = f0 * (1 + 1.4 * np.exp(-t / 0.014))
        if gfrom is not None:
            fg = 440 * 2 ** ((gfrom - 69) / 12); g = int(min(L, 0.35 * sr))
            f[:g] = np.geomspace(fg, f0, g) * (1 + 1.4 * np.exp(-t[:g] / 0.014))
        x = np.sin(2 * np.pi * np.cumsum(f) / sr) * np.exp(-t / decay) * (vel / 127)
        x *= np.minimum(1, t / 0.002) * np.minimum(1, (L - np.arange(L)) / (0.03 * sr))
        y[s:s + L] = x + y[s:s + L] * np.exp(-np.arange(L) / (0.003 * sr))
    y = dsp.sat(y, drive_db)
    y2 = np.stack([y, y], 1)
    y2 = dsp.peq(y2, 88, 1.0, 1.0); y2 = dsp.lp(y2, 2500, order=2)
    return (y2 / (np.abs(y2).max() + 1e-9)).astype(np.float32)


def duck(x, trigger_times, depth_db=-6, att_ms=4, rel_ms=90, sr=SR):
    env = np.ones(len(x))
    a = int(att_ms * 1e-3 * sr); r = int(rel_ms * 1e-3 * sr); d = 10 ** (depth_db / 20)
    for t in trigger_times:
        i = int(t * sr)
        if i >= len(x): continue
        env[i:i + a] = np.minimum(env[i:i + a], np.linspace(1, d, min(a, len(x) - i)))
        j = i + a; seg = np.linspace(d, 1, min(r, max(0, len(x) - j)))
        env[j:j + len(seg)] = np.minimum(env[j:j + len(seg)], seg)
    return x * env[:, None]


def riser(dur_s, sr=SR, seed=5):
    """Band-limited noise sweep 300 → 6000 Hz + rising sine; short, belongs to the drum world."""
    N = int(dur_s * sr); t = np.arange(N) / sr; rng = np.random.default_rng(seed)
    nz = rng.standard_normal(N); out = np.zeros(N); hop = 1024
    for i in range(0, N, hop):
        fc = 300 * (6000 / 300) ** (i / N); seg = nz[i:i + hop]
        sos = ss.butter(2, [fc / 1.6, min(fc * 1.6, sr / 2 - 1)], 'bandpass', fs=sr, output='sos')
        out[i:i + hop] = ss.sosfilt(sos, seg)
    env = (t / dur_s) ** 2.2
    y = out / (np.abs(out).max() + 1e-9) * env
    return np.stack([y, y * 0.9], 1).astype(np.float32)


def impact(sr=SR, seed=6):
    """Sub drop 70 → 32 Hz + saturated noise burst; the 'hit' at a drop."""
    N = int(1.6 * sr); t = np.arange(N) / sr
    f = 70 * (32 / 70) ** np.minimum(1, t / 0.5); sub = np.sin(2 * np.pi * np.cumsum(f) / sr) * np.exp(-t / 0.55)
    nz = dsp.bpf(np.random.default_rng(seed).standard_normal(N), 150, 4000) * np.exp(-t / 0.12)
    y = dsp.sat(sub * 1.0 + nz * 0.35, 6)
    return (np.stack([y, y], 1) / np.abs(y).max()).astype(np.float32)
